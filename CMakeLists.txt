cmake_minimum_required(VERSION 3.16)
project(FinalBuildPower VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(FBS_BUILD_TESTS "Build module tests" ON)
option(FBS_BUILD_EXAMPLES "Build the example" ON)
add_library(fbs_power src/power.c)
add_library(fbs::power ALIAS fbs_power)
set_target_properties(fbs_power PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON EXPORT_NAME power)
target_include_directories(fbs_power PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
  target_link_libraries(fbs_power PUBLIC m)
  target_compile_options(fbs_power PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(FBS_BUILD_TESTS AND BUILD_TESTING)
  add_executable(fbs_test_power tests/test_power.c)
  set_target_properties(fbs_test_power PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_test_power PRIVATE fbs::power)
  if(NOT MSVC)
    target_compile_options(fbs_test_power PRIVATE -UNDEBUG -ffp-contract=off)
  endif()
  add_test(NAME power COMMAND fbs_test_power)
  set_tests_properties(power PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
if(FBS_BUILD_EXAMPLES)
  add_executable(fbs_power_example examples/basic.c)
  set_target_properties(fbs_power_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_power_example PRIVATE fbs::power)
  if(BUILD_TESTING)
    add_test(NAME power_example COMMAND fbs_power_example)
  endif()
endif()
install(TARGETS fbs_power EXPORT FinalBuildPowerTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FinalBuildPowerTargets NAMESPACE fbs::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FinalBuildPower)
install(FILES LICENSE THIRD_PARTY_NOTICES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-power)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-power/upstream OPTIONAL)
